PROPOSED Prisma schema

This is a discussion draft only. It does not replace schema.prisma and has not
been applied to a database. It assumes roles, appointment statuses, and
interaction types are fixed application-wide values. Review existing database
values and plan an append-only migration before adopting these changes.

generator client {
  provider   = "prisma-client-js"
  engineType = "library"
  output     = "../src/generated/prisma"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model Customer {
  id                       String        @id @default(cuid())
  business_id              String
  business                 Business      @relation(fields: [business_id], references: [id], onDelete: Restrict)
  first_name               String
  last_name                String
  phone_number             String
  email                    String
  address                  String
  preferred_contact_method ContactMethod
  createdAt                DateTime      @default(now())
  updatedAt                DateTime      @updatedAt
  notes                    Note[]
  appointments             Appointment[]
  tasks                    Task[]
  interactions             Interaction[]

  @@unique([business_id, phone_number])
  @@unique([business_id, email])
  @@index([business_id, last_name, first_name])
}

model Task {
  id          String     @id @default(cuid())
  customer_id String
  employee_id String
  customer    Customer   @relation(fields: [customer_id], references: [id], onDelete: Restrict)
  employee    Employee   @relation(fields: [employee_id], references: [id], onDelete: Restrict)
  title       String
  description String
  due_date    DateTime
  status      TaskStatus @default(PENDING)
  created_at  DateTime   @default(now())
  updated_at  DateTime   @updatedAt

  @@index([employee_id, due_date])
  @@index([customer_id])
}

model Note {
  id          String   @id @default(cuid())
  customer_id String
  employee_id String
  customer    Customer @relation(fields: [customer_id], references: [id], onDelete: Restrict)
  employee    Employee @relation(fields: [employee_id], references: [id], onDelete: Restrict)
  note_date   DateTime @default(now())
  content     String
  created_at  DateTime @default(now())
  updated_at  DateTime @updatedAt

  @@index([customer_id, note_date])
}

model Employee {
  id           String       @id @default(cuid())
  business_id  String
  user_id      String
  business     Business     @relation(fields: [business_id], references: [id], onDelete: Restrict)
  user         User         @relation(fields: [user_id], references: [id], onDelete: Restrict)
  first_name   String
  last_name    String
  email        String
  phone_number String
  role         EmployeeRole @default(STAFF) @map("role_id")
  is_active    Boolean      @default(true)
  hire_date    DateTime     @db.Date
  created_at   DateTime     @default(now()) @map("create_at")
  appointments Appointment[]
  interactions Interaction[]
  tasks        Task[]
  notes        Note[]

  @@unique([business_id, user_id])
  @@unique([business_id, email])
  @@index([user_id])
  @@index([business_id])
}

model Appointment {
  id                   String                @id @default(cuid())
  customer_id          String
  employee_id          String
  location_id          String
  customer             Customer              @relation(fields: [customer_id], references: [id], onDelete: Restrict)
  employee             Employee              @relation(fields: [employee_id], references: [id], onDelete: Restrict)
  location             Location              @relation(fields: [location_id], references: [id], onDelete: Restrict)
  appointment_date     DateTime
  start_time           DateTime
  end_time             DateTime
  status               AppointmentStatus     @default(SCHEDULED) @map("status_id")
  notes                String
  created_at           DateTime              @default(now())
  updated_at           DateTime              @updatedAt
  appointment_services Appointment_Service[]
  payments             Payment[]

  @@index([customer_id, appointment_date])
  @@index([employee_id, appointment_date])
  @@index([location_id, appointment_date])
}

model Appointment_Service {
  id               String      @id @default(cuid())
  appointment_id   String
  appointment      Appointment @relation(fields: [appointment_id], references: [id], onDelete: Cascade)
  service_id       String
  service          Service     @relation(fields: [service_id], references: [id], onDelete: Restrict)
  quantity         Int
  price_at_booking Decimal

  @@unique([appointment_id, service_id])
  @@index([service_id])
}

model Service {
  id                   String                @id @default(cuid())
  business_id          String
  business             Business              @relation(fields: [business_id], references: [id], onDelete: Restrict)
  name                 String
  category             String
  description          String
  duration_minutes     Int
  base_price           Decimal
  is_Active            Boolean               @default(false)
  appointment_services Appointment_Service[]

  @@unique([business_id, name])
  @@index([business_id])
}

model Interaction {
  id               String          @id @default(cuid())
  customer_id      String
  employee_id      String
  customer         Customer        @relation(fields: [customer_id], references: [id], onDelete: Restrict)
  employee         Employee        @relation(fields: [employee_id], references: [id], onDelete: Restrict)
  type             InteractionType @map("type_id")
  subject          String
  content          String
  interaction_date DateTime        @default(now())
  created_at       DateTime        @default(now())

  @@index([customer_id, interaction_date])
}

model Payment {
  id                    String        @id @default(cuid())
  appointment_id        String
  appointment           Appointment   @relation(fields: [appointment_id], references: [id], onDelete: Restrict)
  payment_method        PaymentMethod
  payment_date          DateTime      @default(now())
  status                PaymentStatus
  transaction_reference String?

  @@index([appointment_id])
}

model Business {
  id            String       @id @default(cuid())
  name          String
  business_type BusinessType
  phone_number  String
  email         String
  created_at    DateTime     @default(now())
  updated_at    DateTime     @updatedAt
  locations     Location[]
  employees     Employee[]
  customers     Customer[]
  services      Service[]
}

model Location {
  id           String        @id @default(cuid())
  business_id  String
  business     Business      @relation(fields: [business_id], references: [id], onDelete: Restrict)
  name         String
  phone_number String?
  email        String?
  address      String
  is_active    Boolean       @default(true)
  created_at   DateTime      @default(now())
  updated_at   DateTime      @updatedAt
  appointments Appointment[]

  @@unique([business_id, name])
  @@unique([business_id, address])
  @@index([business_id])
}

model User {
  id        String     @id @default(cuid())
  username  String     @unique
  createdAt DateTime   @default(now())
  employees Employee[]
}

enum EmployeeRole {
  OWNER
  MANAGER
  STAFF
}

enum AppointmentStatus {
  SCHEDULED
  CONFIRMED
  COMPLETED
  CANCELED
  NO_SHOW
}

enum TaskStatus {
  PENDING
  IN_PROGRESS
  COMPLETED
  CANCELED
}

enum InteractionType {
  CALL
  TEXT
  EMAIL
  DM
}

enum ContactMethod {
  PHONE
  EMAIL
  TEXT
}

enum PaymentMethod {
  CASH
  CARD
  BANK_TRANSFER
}

enum PaymentStatus {
  PENDING
  COMPLETED
  FAILED
  REFUNDED
}

enum BusinessType {
  SPA
  SALON
  CLINIC
  BARBERSHOP
  OTHER
}
